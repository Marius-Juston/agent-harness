import json, os, struct
os.makedirs("/tmp/m", exist_ok=True)
tensors = {"w": [1.0, 2.0, 3.0, 4.0], "b": [0.5, -0.5]}
header, data, off = {}, b"", 0
for k, v in tensors.items():
    raw = struct.pack(f"<{len(v)}f", *v)
    header[k] = {"dtype": "F32", "shape": [len(v)], "data_offsets": [off, off + len(raw)]}
    data += raw; off += len(raw)
h = json.dumps(header).encode()
open("/tmp/m/model.safetensors", "wb").write(struct.pack("<Q", len(h)) + h + data)
json.dump({"model_type": "llama", "architectures": ["LlamaForCausalLM"]}, open("/tmp/m/config.json", "w"))
