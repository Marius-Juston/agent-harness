def mkpdf(path, text):
    stream = f"BT /F1 24 Tf 72 720 Td ({text}) Tj ET".encode()
    objs = [b"<< /Type /Catalog /Pages 2 0 R >>", b"<< /Type /Pages /Kids [3 0 R] /Count 1 >>",
            b"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources << /Font << /F1 5 0 R >> >> /Contents 4 0 R >>",
            b"<< /Length " + str(len(stream)).encode() + b" >>\nstream\n" + stream + b"\nendstream",
            b"<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>"]
    out = bytearray(b"%PDF-1.4\n"); offs = []
    for i, o in enumerate(objs, 1):
        offs.append(len(out)); out += f"{i} 0 obj\n".encode() + o + b"\nendobj\n"
    x = len(out); out += f"xref\n0 {len(objs)+1}\n0000000000 65535 f \n".encode()
    for o in offs: out += f"{o:010d} 00000 n \n".encode()
    out += f"trailer\n<< /Size {len(objs)+1} /Root 1 0 R >>\nstartxref\n{x}\n%%EOF\n".encode()
    open(path, "wb").write(bytes(out))
mkpdf("/tmp/t.pdf", "Hello Gold")
import reportlab, pkgutil
mods = [m.name for m in pkgutil.walk_packages(reportlab.__path__, "reportlab.")]
print(len(mods), [m for m in mods if "pdfbase" in m or "read" in m.lower() or "pars" in m.lower()][:30])
# reportlab writes PDFs; its open-source edition documents no reader. Look for any text-extraction entry point:
# rparsexml and paraparser parse reportlab's own paragraph markup, not PDF files.
hits = [m for m in mods if any(k in m.lower() for k in ("extract", "reader", "pdfrw", "parse")) and m not in ("reportlab.lib.rparsexml", "reportlab.platypus.paraparser")]
print("candidates:", hits)
print("GOLD_NOT" if not hits else "GOLD_CHECK")
