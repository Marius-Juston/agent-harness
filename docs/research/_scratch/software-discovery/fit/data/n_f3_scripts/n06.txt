from capa import compile_to_js

src = (
    "fun add(a: Int, b: Int) -> Int\n"
    "    return a + b\n"
    "\n"
    "fun main()\n"
    "    print(add(2, 3))\n"
)
js = compile_to_js(src)
print(js)
assert isinstance(js, str) and js.strip()
assert "function add(a, b)" in js
assert "return (a + b);" in js or "return a + b;" in js
assert "function main(" in js
assert "add(2, 3)" in js
assert "def " not in js and "fun " not in js
print("FIT_OK")
