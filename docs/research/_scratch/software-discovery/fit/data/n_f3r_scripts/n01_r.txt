import inspect
from multiplayer import Player, User

sig = inspect.signature(Player)
print("Player sig:", sig)
print("Player methods:", [n for n in dir(Player) if not n.startswith("_")])
params = list(sig.parameters.values())
sp = next((p.name for p in params if "static" in p.name.lower()), None)
dp = next((p.name for p in params if "dynamic" in p.name.lower()), None)
assert sp and dp, f"Player has no static/dynamic attribute params: {sig}"
kw = {p.name: "p1" for p in params if p.default is p.empty and p.kind == p.POSITIONAL_OR_KEYWORD and p.name not in (sp, dp)}
kw[sp], kw[dp] = {"team": "red"}, {"score": 0}
p = Player(**kw)

def find(obj, key):
    for v in vars(obj).values():
        if isinstance(v, dict) and key in v:
            return v[key]
    return getattr(obj, key)

assert find(p, "team") == "red"
assert find(p, "score") == 0
setter = next((getattr(p, n) for n in dir(p) if "set" in n.lower() and "dynamic" in n.lower() and callable(getattr(p, n))), None)
print("setter:", setter)
if setter is not None:
    setter("score", 10)
else:
    next(v for v in vars(p).values() if isinstance(v, dict) and "score" in v)["score"] = 10
assert find(p, "score") == 10

u = User("alice", "s3cret")
print("User methods:", [n for n in dir(u) if not n.startswith("_")])
assert getattr(u, "username", None) == "alice"
check = next((getattr(u, n) for n in dir(u) if ("check" in n.lower() or "verify" in n.lower()) and callable(getattr(u, n))), None)
assert check is not None, "User has no password check method"
assert check("s3cret") is True
assert not check("wrong")
print("FIT_OK")
