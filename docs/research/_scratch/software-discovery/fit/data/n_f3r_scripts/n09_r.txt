import os, subprocess, tempfile, inspect
import ezgb

print("ezgb exports:", [n for n in dir(ezgb) if not n.startswith("_")])
repo = tempfile.mkdtemp(dir="/tmp")
os.environ.update(HOME=repo, GIT_AUTHOR_NAME="T", GIT_AUTHOR_EMAIL="t@e.x",
                  GIT_COMMITTER_NAME="T", GIT_COMMITTER_EMAIL="t@e.x")
subprocess.run(["git", "init", "-q", repo], check=True)
subprocess.run(["git", "-C", repo, "config", "user.name", "T"], check=True)
subprocess.run(["git", "-C", repo, "config", "user.email", "t@e.x"], check=True)

cands = [getattr(ezgb, n) for n in dir(ezgb) if inspect.isclass(getattr(ezgb, n))]
Cls = next(c for c in cands if any("create" in m and "bug" in m.lower() for m in dir(c)))
print("using", Cls, inspect.signature(Cls))
meths = [m for m in dir(Cls) if not m.startswith("_")]
print("methods:", meths)
create = next(m for m in meths if "create" in m and "bug" in m)
addc = next(m for m in meths if "comment" in m and ("add" in m or "create" in m))
get = next(m for m in meths if m.startswith("get") and "bug" in m and "bugs" not in m)

gb = Cls(repo)
title, body, comment = "Fit test bug", "Initial description", "A follow-up comment"
bug = getattr(gb, create)(title, body)
bug_id = bug if isinstance(bug, str) else (bug.get("id") if isinstance(bug, dict) else getattr(bug, "id", None) or getattr(bug, "human_id"))
assert bug_id, "no bug id returned"
getattr(gb, addc)(bug_id, comment)

text = repr(getattr(gb, get)(bug_id))
if hasattr(gb, "stop"):
    gb.stop()
assert title in text, "title not found on fetched bug"
assert body in text, "initial message not found on fetched bug"
assert comment in text, "added comment not found on fetched bug"
print("FIT_OK")
