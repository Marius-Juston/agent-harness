import asyncio
import inspect
import threading
from http.server import BaseHTTPRequestHandler, HTTPServer

import requests


class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        body = b"hello-async"
        self.send_response(200)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *args):
        pass


server = HTTPServer(("127.0.0.1", 0), Handler)
threading.Thread(target=server.serve_forever, daemon=True).start()
url = f"http://127.0.0.1:{server.server_address[1]}/"


async def main():
    # An asyncio-native client must expose an awaitable request API.
    assert inspect.iscoroutinefunction(requests.get), "requests.get is not a coroutine function"
    pending = requests.get(url, timeout=5)
    assert inspect.isawaitable(pending), "requests.get did not return an awaitable"
    resp = await pending
    assert resp.status_code == 200
    assert resp.text == "hello-async"


asyncio.run(main())
server.shutdown()
print("FIT_OK")
