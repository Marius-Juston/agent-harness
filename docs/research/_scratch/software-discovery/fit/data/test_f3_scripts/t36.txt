import socket

def _no_network(*args, **kwargs):
    raise RuntimeError("network access attempted")

socket.socket.connect = _no_network
socket.create_connection = _no_network

import pysbd

seg = pysbd.Segmenter(language="en", clean=False)

def split(text):
    return [s.strip() for s in seg.segment(text) if s.strip()]

# Basic period, question, exclamation boundaries
assert split("Hello World. My name is Jonas.") == ["Hello World.", "My name is Jonas."]
assert split("What is your name? My name is Jonas.") == ["What is your name?", "My name is Jonas."]
assert split("There it is! I found it.") == ["There it is!", "I found it."]

# Abbreviations, initials and decimals must not cause splits
assert split("My name is Jonas E. Smith.") == ["My name is Jonas E. Smith."]
assert split("Please turn to p. 55.") == ["Please turn to p. 55."]
assert split("I have a 3.5 GPA.") == ["I have a 3.5 GPA."]

text = ("Mr. Smith went to Washington. He paid $3.50 for coffee! "
        "Did he like it? Yes.")
result = split(text)
assert result == [
    "Mr. Smith went to Washington.",
    "He paid $3.50 for coffee!",
    "Did he like it?",
    "Yes.",
], result

# Character spans map back onto the original text
spans = pysbd.Segmenter(language="en", clean=False, char_span=True).segment(text)
assert len(spans) == 4
for span in spans:
    assert text[span.start:span.end] == span.sent
    assert span.sent.strip() in result

print("FIT_OK")
