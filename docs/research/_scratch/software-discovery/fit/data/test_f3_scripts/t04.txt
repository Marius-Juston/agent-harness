import io
from pathlib import Path
from ruamel.yaml import YAML

src = """\
# Top-level config comment
server:
  host: localhost  # the hostname
  port: 8080       # listening port
# Section about users
users:
  - alice  # admin
  - bob
"""

path = Path("/tmp/fit_test.yaml")
path.write_text(src)

yaml = YAML()  # round-trip mode by default
data = yaml.load(path)
data["server"]["port"] = 9090
with path.open("w") as f:
    yaml.dump(data, f)

out = path.read_text()
print(out)

assert "port: 9090" in out
assert "8080" not in out
assert "# Top-level config comment" in out
assert "# the hostname" in out
assert "# listening port" in out
assert "# Section about users" in out
assert "# admin" in out
assert "host: localhost" in out
# unchanged lines are byte-identical
assert out.replace("9090", "8080") == src

reloaded = YAML().load(io.StringIO(out))
assert reloaded["server"]["port"] == 9090
assert list(reloaded["users"]) == ["alice", "bob"]
print("FIT_OK")
